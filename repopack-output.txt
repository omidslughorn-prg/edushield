# دیکشنری ترجمه وضعیت‌ها به فارسی جهت درج در پیامک
STATUS_TRANSLATIONS = {
    AttendanceStatus.PRESENT: "حاضر",
    AttendanceStatus.LATE: "تاخیر ورود",
    AttendanceStatus.EXCUSED_ABSENCE: "غیبت موجه",
    AttendanceStatus.UNEXCUSED_ABSENCE: "غیبت غیرموجه",
}

@router.post("/record", response_model=AttendanceResponseSchema, status_code=status.HTTP_201_CREATED)
async def record_attendance(
    data: AttendanceCreateSchema,
    background_tasks: BackgroundTasks,
    db: AsyncSession = Depends(get_db)
):
    """ثبت سریع تاخیر یا غیبت دانش‌آموز به همراه ارسال پیامک خودکار به ولی"""
    
    # ۱. بررسی وجود دانش‌آموز
    student = await db.get(Student, data.student_id)
    if not student:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"دانش‌آموزی با شناسه {data.student_id} یافت نشد."
        )

    # ۲. ایجاد رکورد حضور و غیاب در دیتابیس
    record_id = f"ATT-{date.today().strftime('%Y%m%d')}-{uuid.uuid4().hex[:6]}"
    
    db_record = AttendanceRecord(
        id=record_id,
        student_id=data.student_id,
        record_date=data.record_date,
        entry_time=data.entry_time,
        delay_minutes=str(data.delay_minutes) if data.delay_minutes is not None else None,
        status=data.status,
        description=data.description
    )

    db.add(db_record)
    await db.commit()
    await db.refresh(db_record)

    # ۳. افزودن وظیفه ارسال پیامک به BackgroundTasks (بدون معطل کردن کاربر)
    if student.guardian_mobile:
        student_fullname = f"{student.first_name} {student.last_name}"
        status_text = STATUS_TRANSLATIONS.get(data.status, "ثبت شده")
        time_str = data.entry_time.strftime("%H:%M") if data.entry_time else None

        background_tasks.add_task(
            sms_service.send_attendance_notification,
            mobile=student.guardian_mobile,
            student_name=student_fullname,
            status_text=status_text,
            record_date=str(data.record_date),
            entry_time=time_str,
            delay_minutes=str(data.delay_minutes) if data.delay_minutes else None
        )

    return db_record


@router.get("/today", response_model=List[AttendanceResponseSchema])

        plain_otp=verify_data.otp_code,

        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="کد تایید واردشده اشتباه یا منقضی شده است.",
        )

    return {
        "status": "success",
        "message": "کد OTP با موفقیت تایید شد.",
        "verified": True,
    }

import hashlib
import random
import string


def generate_otp_code(length: int = 6) -> str:
    """تولید یک کد OTP ۶ رقمی عددی"""
    return "".join(random.choices(string.digits, k=length))


def hash_otp_code(otp_code: str) -> str:
    """هش‌سازی کد OTP جهت ذخیره یا مقایسه امن در دیتابیس (عدم ذخیره کد متنی)"""
    return hashlib.sha256(otp_code.encode("utf-8")).hexdigest()


def verify_otp_code(plain_otp: str, hashed_otp: str) -> bool:
    """اعتبارسنجی کد واردشده توسط ولی با هش ذخیره‌شده"""
    return hash_otp_code(plain_otp) == hashed_otp

import hashlib
import json
from datetime import datetime


def generate_legal_transaction_hash(
    transaction_id: str,
    student_id: str,
    guardian_national_id: str,
    guardian_mobile: str,
    reason: str,
    timestamp_utc: datetime,
    otp_code_hash: str,
    signature_vector: str,
) -> str:
    """تولید هش SHA-256 از تمام داده‌های تراکنش مرخصی جهت قفل رمزنگاری‌شده و اثبات اصالت در دادگاه (ماده ۱۵ قانون تجارت الکترونیک)"""
    payload = {
        "transaction_id": transaction_id,
        "student_id": student_id,
        "guardian_national_id": guardian_national_id,
        "guardian_mobile": guardian_mobile,
        "reason": reason,
        "timestamp_utc": timestamp_utc.isoformat(),
        "otp_code_hash": otp_code_hash,
        "signature_vector": signature_vector,
    }

    # تبدیل دیکشنری به رشته استاندارد JSON با ترتیب ثابت کلیدها جهت یکسان بودن هش
    payload_string = json.dumps(payload, sort_keys=True)

    # تولید هش SHA-256
    return hashlib.sha256(payload_string.encode("utf-8")).hexdigest()

import os
import logging
from typing import Optional
from kavenegar import KavenegarAPI, APIException, HTTPException

# تنظیمات لوگ برای مشاهده پیامک‌های ارسالی در محیط توسعه
logger = logging.getLogger("sms_service")
logging.basicConfig(level=logging.INFO)

# دریافت کلید API کاوه‌نگار از فایل تنظیمات .env
KAVENEGAR_API_KEY = os.getenv("KAVENEGAR_API_KEY", "your_api_key_here")

class SMSService:
    """ سرویس ارسال پیامک اطلاع‌رسانی به اولیا """

    @staticmethod
    async def send_attendance_notification(
        mobile: str,
        student_name: str,
        status_text: str,
        record_date: str,
        entry_time: Optional[str] = None,
        delay_minutes: Optional[str] = None
    ) -> bool:
        """ ارسال پیامک اطلاع‌رسانی تاخیر/غیبت به ولی دانش‌آموز """

        # ۱. تنظیم متن پیامک
        if entry_time and delay_minutes and delay_minutes != "0":
            message = (
                f"ولی محترم، فرزند شما {student_name} در تاریخ {record_date} "
                f"ساعت {entry_time} با {delay_minutes} دقیقه تاخیر وارد مدرسه شد.\n"
                f"سامانه ناظم‌همراه"
            )
        else:
            message = (
                f"ولی محترم، وضعیت حضور فرزند شما {student_name} در تاریخ {record_date} "
                f"به صورت «{status_text}» ثبت گردید.\n"
                f"سامانه ناظم‌همراه"
            )

        # ۲. حالت شبیه‌سازی (وقتی کلید واقعی در .env تنظیم نشده است)
        if not KAVENEGAR_API_KEY or KAVENEGAR_API_KEY == "your_api_key_here":
            logger.info("=" * 60)
            logger.info(f"📲 [KAVENEGAR MOCK] To: {mobile}")
            logger.info(f"✉️ [MESSAGE]:\n{message}")
            logger.info("=" * 60)
            return True

        # ۳. حالت ارسال واقعی از طریق API کاوه‌نگار
        try:
            api = KavenegarAPI(KAVENEGAR_API_KEY)
            params = {
                'receptor': mobile,
                'message': message,
            }
            response = api.sms_send(params)
            logger.info(f"✅ پیامک واقعی با موفقیت ارسال شد: {response}")
            return True
        except (APIException, HTTPException) as e:
            logger.error(f"❌ خطای ارسال پیامک کاوه‌نگار: {e}")
            return False

sms_service = SMSService()

import io
import pandas as pd
from datetime import date
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from app.db.models import AttendanceRecord, Student, ClassRoom

class ExcelExportService:
    """سرویس تولید گزارش‌های اکسل جهت سامانه سیدا و بایگانی مدرسه"""

    @staticmethod
    async def generate_attendance_excel(
        db: AsyncSession, 
        start_date: date, 
        end_date: date
    ) -> io.BytesIO:
        """استخراج گزارش تاخیرها و غیبت‌ها در بازه زمانی مشخص به فرمت اکسل"""
        
        # ۱. کوئری دریافت اطلاعات حضور/غیبت به همراه اطلاعات دانش‌آموز و کلاس
        query = (
            select(
                AttendanceRecord.record_date,
                Student.national_id,
                Student.first_name,
                Student.last_name,
                ClassRoom.name.label("class_name"),
                AttendanceRecord.status,
                AttendanceRecord.entry_time,
                AttendanceRecord.delay_minutes,
                AttendanceRecord.description
            )
            .join(Student, AttendanceRecord.student_id == Student.id)
            .join(ClassRoom, Student.class_id == ClassRoom.id)
            .where(AttendanceRecord.record_date.between(start_date, end_date))
            .order_by(AttendanceRecord.record_date.desc())
        )

        result = await db.execute(query)
        rows = result.all()

        # ۲. تبدیل داده‌ها به لیست دیکشنری جهت بارگذاری در Pandas
        data_list = []
        status_farsi = {
            "present": "حاضر",
            "late": "تاخیر ورود",
            "excused": "غیبت موجه",
            "unexcused": "غیبت غیرموجه"
        }

        for row in rows:
            data_list.append({
                "تاریخ ثبت": str(row.record_date),
                "کد ملی دانش‌آموز": row.national_id,
                "نام": row.first_name,
                "نام خانوادگی": row.last_name,
                "کلاس": row.class_name,
                "وضعیت حضور": status_farsi.get(row.status.value, str(row.status)),
                "زمان ورود": str(row.entry_time) if row.entry_time else "-",
                "میزان تاخیر (دقیقه)": row.delay_minutes if row.delay_minutes else "0",
                "توضیحات / علت": row.description if row.description else "-"
            })

        # ۳. ساخت DataFrame با Pandas
        df = pd.DataFrame(data_list)

        # اگر داده‌ای در این بازه وجود نداشت، یک جدول خالی با ستون‌ها می‌سازد
        if df.empty:
            df = pd.DataFrame(columns=[
                "تاریخ ثبت", "کد ملی دانش‌آموز", "نام", "نام خانوادگی",
                "کلاس", "وضعیت حضور", "زمان ورود", "میزان تاخیر (دقیقه)", "توضیحات / علت"
            ])

        # ۴. ذخیره در حافظه رم (BytesIO) به جای ذخیره روی دیسک
        output = io.BytesIO()
        with pd.ExcelWriter(output, engine='openpyxl') as writer:
            df.to_excel(writer, index=False, sheet_name='گزارش_حضور_غیبت')

        output.seek(0)
        return output

export_service = ExcelExportService()

                    status === 'unexcused'
                      ? 'bg-red-500 text-white border-red-500 shadow-md'
                      : 'border-slate-200 text-slate-600 hover:bg-slate-50'
                  }`}
                >
                  <UserX className="w-6 h-6" />
                  <span>غیبت غیرموجه</span>
                </button>

                {/* Excused */}
                <button
                  type="button"
                  onClick={() => setStatus('excused')}
                  className={`p-4 rounded-xl border flex flex-col items-center justify-center gap-2 font-medium text-sm transition-all ${
                    status === 'excused'
                      ? 'bg-blue-500 text-white border-blue-500 shadow-md'
                      : 'border-slate-200 text-slate-600 hover:bg-slate-50'
                  }`}
                >
                  <UserCheck className="w-6 h-6" />
                  <span>غیبت موجه</span>
                </button>
              </div>
            </div>

            {/* Late-specific inputs */}
            {status === 'late' && (
              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                {/* Entry Time */}
                <div>
                  <label
                    htmlFor="entryTime"
                    className="block mb-2 font-medium"
                  >
                    زمان ورود
                  </label>

                  <input
                    id="entryTime"
                    type="time"
                    value={entryTime}
                    onChange={(e) => setEntryTime(e.target.value)}
                    className="w-full p-3 border border-slate-300 rounded-xl focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none"
                    required
                  />
                </div>

                {/* Delay */}
                <div>
                  <label
                    htmlFor="delayMinutes"
                    className="block mb-2 font-medium"
                  >
                    میزان تاخیر (دقیقه)
                  </label>

                  <input
                    id="delayMinutes"
                    type="number"
                    min="0"
                    value={delayMinutes}
                    onChange={(e) => setDelayMinutes(e.target.value)}
                    className="w-full p-3 border border-slate-300 rounded-xl focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none"
                    required
                  />
                </div>
              </div>
            )}

            {/* Description */}
            <div>
              <label
                htmlFor="description"
                className="block mb-2 font-medium"
              >
                توضیحات / علت (اختیاری)
              </label>

              <textarea
                id="description"
                value={description}
                onChange={(e) => setDescription(e.target.value)}
                rows="3"
                className="w-full p-3 border border-slate-300 rounded-xl focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none resize-none"
                placeholder="توضیح مختصر در صورت نیاز..."
              />
            </div>

            {/* Submit */}
            <button
              type="submit"
              disabled={loading}
              className="w-full bg-blue-600 hover:bg-blue-700 disabled:bg-slate-400 disabled:cursor-not-allowed text-white font-bold py-3 px-5 rounded-xl transition-colors"
            >
              {loading
                ? 'در حال ثبت...'
                : 'ثبت و ارسال پیامک خودکار به ولی'}
            </button>
          </form>
        </section>

        {/* Today Records List */}
        <section className="bg-white rounded-2xl shadow-sm border border-slate-200 p-5 md:p-6">
          <div className="flex justify-between items-center mb-4">
            <h3 className="text-lg font-bold">
              لیست ثبت‌شده‌های امروز
            </h3>
            <button
              type="button"
              onClick={handleDownloadExcel}
              className="bg-emerald-600 hover:bg-emerald-700 text-white px-3 py-1.5 rounded-xl text-sm font-medium flex items-center gap-2 transition-colors"
            >
              📊 دریافت خروجی اکسل
            </button>
          </div>

          {records.length === 0 ? (
            <div className="text-center py-8 text-slate-500">
              هیچ رکوردی برای امروز ثبت نشده است.
            </div>
          ) : (
            <div className="space-y-3">
              {records.map((rec, index) => (
                <div
                  key={rec.id ?? `${rec.student_id}-${index}`}
                  className="flex flex-col md:flex-row md:items-center md:justify-between gap-3 p-4 rounded-xl bg-slate-50 border border-slate-200"
                >
                  <div>
                    <div className="font-bold text-slate-800">
                      {rec.student_id}
                    </div>

                    <div className="text-sm text-slate-500 mt-1">
                      زمان ورود: {rec.entry_time || '-'} | تاخیر:{' '}
                      {rec.delay_minutes || 0} دقیقه
                    </div>
                  </div>

                  <div
                    className={`inline-flex self-start md:self-auto px-3 py-1 rounded-full text-sm font-medium ${
                      rec.status === 'late'
                        ? 'bg-amber-100 text-amber-700'
                        : rec.status === 'unexcused'
                        ? 'bg-red-100 text-red-700'
                        : 'bg-blue-100 text-blue-700'
                    }`}
                  >
                    {rec.status === 'late'
                      ? 'تاخیر ورود'
                      : rec.status === 'unexcused'
                      ? 'غیبت غیرموجه'
                      : 'غیبت موجه'}
                  </div>
                </div>
              ))}
            </div>
          )}
        </section>
      </main>
    </div>
  );
}
